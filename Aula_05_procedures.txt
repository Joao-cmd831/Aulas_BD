drop procedure if exists listar_paises $$
create procedure listar_paises (in id int)
begin
	if (id is null) then
	select * from pais;
else
select * from pais where id_pais = id;
end if;
end $$

create database biblio;
use biblio;
drop table biblio . livro;

create table biblio . livro(
id_livro int not null primary key auto_increment,
titulo_livro varchar(100)not null
) engine = InnoDB;

create table biblio . usuario (
id_usuario int not null primary key auto_increment,
nome_usuario varchar(100) not null,
email_usuario varchar(250),
data_nasc_usuario date,
quant_emprest_usuario int not null default 0
)engine = InnoDB;


create table biblio . emprestimo (
id_emprestimo int not null primary key auto_increment,
usuario_id_usuario int not null,
livro_id_livro int not null,
data_emprestimo date not null,
data_devolucao date not null,
data_entrega date,
foreign key(usuario_id_usuario) references usuario(id_usuario),
foreign key(livro_id_livro) references livro(id_livro));
